% PRINCIPALS

Principal(board).
Principal(procurement).
Principal(facility).
Principal(hiring).
Principal(auditor_a).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% PREDICATE CLASSES
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% ------------------------------------------------------------
% Source facts
% ------------------------------------------------------------

% Semantic label attached by the source application/agent when
% a source artifact is created.
%
% Meaning:
%   receiving M counts as acquiring component C of project P.
source_fact Carries(M, P, C).


% ------------------------------------------------------------
% Runtime facts
% ------------------------------------------------------------

% X was exposed to the computation that produced Y.
% Generated by the computation monitor.
runtime_fact DerivedFrom(Y, X).

% M was successfully delivered to A.
% Generated by the messaging runtime after commit.
runtime_fact Received(A, M).

% M is currently being considered for delivery to A.
% Generated temporarily during receive-time checking.
runtime_fact Incoming(A, M).


% ------------------------------------------------------------
% Derived facts
% ------------------------------------------------------------

% A has accumulated component C of project P.
derived_fact Knows(A, P, C).

% Policy decision for an attempted event.
derived_fact Deny(Event).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% STORAGE / PLACEMENT
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Each principal A persistently stores only its own knowledge state.
persistent_local(A):
    Knows(A, _, _).

% Received history may also be retained locally.
persistent_local(A):
    Received(A, _).

% Computation provenance is local to the producing principal.
local_runtime(A):
    DerivedFrom(_, _).

% Incoming facts exist only inside one receive transaction.
ephemeral_local(A):
    Incoming(A, _).

% Carries facts are bound to artifacts/messages.
artifact_local(M):
    Carries(M, _, _).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% TRANSPORT
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Only Carries facts associated with message M cross principals.
export_with_message(M):
    Carries(M, _, _).

% The following never travel with a message.
do_not_export:
    Knows(_, _, _).
    Received(_, _).
    DerivedFrom(_, _).
    Incoming(_, _).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% RULES
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%


% R1.
% Restricted components conservatively propagate through computation.
Carries(Y, P, C) :-
    DerivedFrom(Y, X),
    Carries(X, P, C).


% R2.
% Successful receipt adds the carried component to receiver knowledge.
Knows(A, P, C) :-
    Received(A, M),
    Carries(M, P, C).


% R3.
% A non-Board principal cannot receive a third distinct component
% of the same project.
Deny(receive(A, M)) :-
    A != board,
    Incoming(A, M),
    Carries(M, P, C3),
    Knows(A, P, C1),
    Knows(A, P, C2),
    C1 != C2,
    C1 != C3,
    C2 != C3.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% SOURCE CREATION PROTOCOL
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% When the source application creates an unrestricted message:
create_source(M, Payload):
    create_artifact(M, Payload).

% Example:
create_source(budget_p,
              "Next-quarter procurement budget: $8M").


% When the source application creates a restricted message:
create_restricted_source(M, Payload, P, C):
    create_artifact(M, Payload),
    assert Carries(M, P, C).

% Examples:
create_restricted_source(
    reason_p,
    "Nova will terminate Supplier X",
    nova,
    procurement).

create_restricted_source(
    reason_f,
    "Nova will consolidate Building 4",
    nova,
    facility).

create_restricted_source(
    reason_h,
    "Nova will freeze Team Z hiring",
    nova,
    hiring).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% COMPUTATION PROTOCOL
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% A computation executes at principal A.
%
% Inputs = all artifacts exposed to the invocation.
% Y      = produced output artifact.

compute(A, Inputs, Y):

    for each X in Inputs:
        assert_local(A, DerivedFrom(Y, X)).

    run_local_closure(A).

    materialize_on_artifact(Y):
        Carries(Y, _, _).


% Example:
%
% Inputs:
%   reason_p
%
% Output:
%   summary_p
%
% Runtime:
%
%   DerivedFrom(summary_p, reason_p)
%
% Rule R1 derives:
%
%   Carries(summary_p, nova, procurement)


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% SEND PROTOCOL
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

send(Sender, Receiver, M):

    Envelope :=
        payload(M)
        +
        {
            Carries(M, P, C)
            | Carries(M, P, C) holds
        }.

    transmit(Sender, Receiver, Envelope).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% RECEIVE PROTOCOL
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

receive(A, Envelope(M, TransportedFacts)):

    begin_ephemeral_context E.

    assert_E Incoming(A, M).

    for each transported fact:
        assert_E Carries(M, P, C).

    evaluate:
        Deny(receive(A, M))
    over:
        LocalStore(A) union E.

    if Deny(receive(A, M)):
        discard E.
        reject M.

    else:
        deliver M.

        commit_local(A, Received(A, M)).

        import_artifact_facts(M):
            Carries(M, _, _).

        run_local_closure(A).

        persist:
            Knows(A, _, _).

        discard E.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% DENY TRANSACTION SEMANTICS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Failed receive attempts do not mutate persistent state.

Deny(receive(A, M))
    =>
    not commit Received(A, M),
    not derive new Knows(A, _, _),
    discard Incoming(A, M).


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% ALLOW TRANSACTION SEMANTICS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Allow(receive(A, M))
    =>
    commit Received(A, M),
    retain Carries(M, _, _),
    derive Knows(A, _, _).

LocalStore(procurement) = {

LocalStore(auditor_a) = {
    Knows(auditor_a, nova, procurement),

    ...
}.

% EXAMPLE DENIED RECEIVE
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Incoming message:
Carries(reason_h, nova, hiring).

% Auditor local state:
Knows(auditor_a, nova, procurement).
Knows(auditor_a, nova, facility).

% Ephemeral receive fact:
Incoming(auditor_a, reason_h).

% R3 derives:
Deny(receive(auditor_a, reason_h)).

% Therefore:
%
%   reason_h is not delivered
%   Received(auditor_a, reason_h) is not committed
%   Knows(auditor_a, nova, hiring) is not created
%
